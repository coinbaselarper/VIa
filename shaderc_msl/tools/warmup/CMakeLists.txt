cmake_minimum_required(VERSION 3.24)
project(shaderc_msl_warmup LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "macOS only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(GTA6RT_METAL_TESTS OFF)
add_subdirectory(${_root}/port/metal metal)
add_subdirectory(${_root}/port/shaderc_msl shaderc_msl)
add_executable(shaderc_msl_warmup warmup_build.cpp)
target_link_libraries(shaderc_msl_warmup PRIVATE gta6rt_metal gta6rt_shaderc_msl)
